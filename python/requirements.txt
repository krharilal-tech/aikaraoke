# AI Karaoke Maker — Python worker dependencies.
# Install into python/venv (see python/README.md):
#   python -m venv venv
#   venv\Scripts\pip install -r requirements.txt      (Windows)
#   venv/bin/pip install -r requirements.txt           (Linux/Ubuntu)

# --- YouTube download (Downloading stage) ---
yt-dlp>=2024.8.6

# --- HTTP calls: LRCLIB/Musixmatch/Genius lookups, OpenAI API (Extracting
#     Lyrics / Generating AI Backgrounds stages) ---
requests>=2.32.0
openai>=1.40.0

# --- Vocal/instrumental separation (Removing Vocals stage) ---
# numpy is pinned explicitly: on some platforms pip's resolver installs
# demucs without pulling in numpy as a transitive dependency, which breaks
# torch/demucs at import time ("ModuleNotFoundError: No module named 'numpy'").
numpy>=1.26.0
demucs>=4.0.1

# torch/torchaudio and the pyannote/lightning/omegaconf stack below are all
# pinned to exact versions, not just floors — confirmed together (in a
# from-scratch isolated venv, matching this app's real WhisperX call
# exactly) to load pyannote.audio's VAD checkpoint cleanly under PyTorch
# 2.6+'s weights_only=True default. Loosening any one of these is what
# broke this before: pyannote-audio pulls in torchcodec, which pins torch
# tightly enough that pip's resolver will silently swap in a different,
# unverified torch build to satisfy it if torch itself isn't pinned here
# too — see services/whisperx_engine.py's
# _allow_legacy_torch_checkpoint_loading() for the weights_only fix itself,
# and the Dockerfile for why torch is no longer installed as a separate
# step ahead of this file.
torch==2.8.0
torchaudio==2.8.0
pyannote-audio==4.0.7
pytorch-lightning==2.6.5
lightning==2.6.5
omegaconf==2.3.1

# --- Lyric transcription + forced alignment (Extracting Lyrics /
#     Synchronizing Lyrics stages, used when no lyrics DB has a match) ---
whisperx==3.8.6

# transformers is a whisperx transitive dependency already, but is pinned
# explicitly here because services/malayalam_asr.py imports it directly —
# generic multilingual Whisper cannot render correct Malayalam script (see
# docs/TROUBLESHOOTING.md), so Malayalam transcription loads a
# Malayalam-finetuned checkpoint via transformers directly instead.
transformers>=4.40.0

# --- RunPod Serverless GPU worker (handler.py) ---
# Only actually used when running inside the RunPod container — harmless,
# lightweight extra install in a local WAMP/Ubuntu venv that never imports
# it. Kept in the same requirements.txt rather than a separate file so the
# container and local dev installs can't drift out of sync on every other
# dependency they do share.
runpod>=1.7.0

# runpod/base's image sets HF_HUB_ENABLE_HF_TRANSFER=1 by default (faster
# HuggingFace downloads), but huggingface_hub only actually uses that
# accelerated path if the hf_transfer package is separately installed —
# without it, every huggingface_hub download (whisperx/pyannote's VAD and
# alignment models, the Malayalam checkpoint) fails outright rather than
# silently falling back to the normal downloader.
hf_transfer>=0.1.6
